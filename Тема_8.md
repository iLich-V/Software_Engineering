```python
# Определение базового класса Shape
class Shape:
    # Метод area (площадь) - должен быть переопределен в дочерних классах
    def area(self):
        pass  # Метод не реализован в базовом классе

# Определение класса Rectangle, наследующего от Shape
class Rectangle(Shape):
    # Конструктор класса Rectangle с параметрами ширины и высоты
    def __init__(self, width, height):
        # Инициализация атрибута ширины прямоугольника
        self.width = width
        # Инициализация атрибута высоты прямоугольника
        self.height = height

    # Переопределение метода area для вычисления площади прямоугольника
    def area(self):
        # Возвращает площадь по формуле
        return self.width * self.height

# Определение класса Circle, наследующего от Shape
class Circle(Shape):
    # Конструктор класса Circle с параметром радиуса
    def __init__(self, radius):
        # Инициализация атрибута радиуса круга
        self.radius = radius

    # Переопределение метода area для вычисления площади круга
    def area(self):
        # Возвращает площадь по формуле
        return 3.14 * self.radius * self.radius

# Создание экземпляра класса Rectangle с шириной 4 и высотой 5
rect = Rectangle(4, 5)
# Создание экземпляра класса Circle с радиусом 3
circle = Circle(3)

# Вывод площади прямоугольника в консоль
print(rect.area())
# Вывод площади круга в консоль
print(circle.area())
```
