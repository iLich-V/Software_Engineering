```python
def longest_words(file):
    with open(file, encoding='utf-8') as f:
        words = f.read().split()
        max_length = len(max(words, key=len))
        sought_words = []
        for word in words:
            if len(word) == max_length:
                sought_words.append(word)

        if len(sought_words) == 1:
            return sought_words[0]
        return sought_words


print(longest_words('input.txt'))
```

## Самостоятельная работа №1
Найдите в интернете любую статью (объем статьи не менее 200 слов), скопируйте ее содержимое в файл и напишите программу, которая считает количество 
слов в текстовом файле и определит самое часто встречающееся слово. Результатом выполнения задачи будет: скриншот файла со статьей, листинг кода, и 
вывод в консоль, в котором будет указана вся необходимая информация.

```python
from collections import Counter
import re

def count_words(filename):
    try:
        with open(filename, 'r', encoding='utf-8') as file:
            text = file.read()

            words = re.findall(r'\b[а-яa-z]+\b', text.lower())

            total_words = len(words)

            if words:
                word_counts = Counter(words)
                most_common_word, count = word_counts.most_common(1)[0]
            else:
                most_common_word, count = None, 0

            return total_words, most_common_word, count

    except FileNotFoundError:
        print(f"Ошибка: Файл '{filename}' не найден")
        return 0, None, 0
    except Exception as e:
        print(f"Ошибка при чтении файла: {e}")
        return 0, None, 0

if __name__ == "__main__":
    filename = input("Введите имя файла: ")

    total, most_common, frequency = count_words(filename)

    if total > 0:
        print(f"\nРезультаты анализа файла '{filename}':")
        print(f"Общее количество слов: {total}")
        if most_common:
            print(f"Самое частое слово: '{most_common}'")
            print(f"Оно встречается {frequency} раз(а)")
        else:
            print("В файле нет слов для анализа")
    else:
        print("Не удалось проанализировать файл")
```

```python
import json
from datetime import datetime

def add_expense():
    try:
        amount = float(input("Введите сумму расхода: "))
        category = input("Введите категорию: ")
        description = input("Введите описание: ")

        expense = {
            'date': datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
            'amount': amount,
            'category': category,
            'description': description
        }

        try:
            with open('expenses.json', 'r', encoding='utf-8') as file:
                expenses = json.load(file)
        except FileNotFoundError:
            expenses = []

        expenses.append(expense)

        with open('expenses.json', 'w', encoding='utf-8') as file:
            json.dump(expenses, file, ensure_ascii=False, indent=2)

        print("Расход успешно добавлен")

    except ValueError:
        print("Ошибка: сумма должна быть числом")
    except Exception as e:
        print(f"Произошла ошибка: {e}")

def show_expenses():
    try:
        with open('expenses.json', 'r', encoding='utf-8') as file:
            expenses = json.load(file)

        if not expenses:
            print("Расходы не найдены")
            return

        print("\nВаша книга расходов")
        print("=" * 50)

        total = 0
        for i, expense in enumerate(expenses, 1):
            print(f"{i}. {expense['date']}")
            print(f"   Сумма: {expense['amount']} руб.")
            print(f"   Категория: {expense['category']}")
            print(f"   Описание: {expense['description']}")
            print("-" * 30)
            total += expense['amount']

        print(f"Общая сумма: {total} руб.")

    except FileNotFoundError:
        print("Файл с расходами не найден")
    except Exception as e:
        print(f"Ошибка при чтении файла: {e}")

def main():
    while True:
        print("\nКнига расходов")
        print("1. Добавить расход")
        print("2. Показать все расходы")
        print("3. Выйти")

        choice = input("Выберите действие: ")

        if choice == '1':
            add_expense()
        elif choice == '2':
            show_expenses()
        elif choice == '3':
            print("До свидания!")
            break
        else:
            print("Неверный выбор, попробуйте снова")

if __name__ == "__main__":
    main()
```

## Самостоятельная работа №3
Имеется файл input.txt с текстом на латинице. Напишите программу, которая выводит следующую статистику по тексту: количество букв латинского алфавита; число слов; число строк.

Текст в файле:
Beautiful is better than ugly.
Explicit is better than implicit.
Simple is better than complex.
Complex is better than complicated.

Ожидаемый результат:
Input file contains:
108 letter
20 words
4 lines

```python
def analyze_file(filename):
    try:
        with open(filename, 'r', encoding='utf-8') as file:
            content = file.read()

        lines = content.split('\n')
        lines = [line for line in lines if line.strip()]
        num_lines = len(lines)

        words = content.split()
        num_words = len(words)

        num_letters = 0
        for char in content:
            if char.isalpha() and char.isascii():
                num_letters += 1

        print(f"Input file contains:")
        print(f"{num_letters} letter{'s' if num_letters != 1 else ''}")
        print(f"{num_words} word{'s' if num_words != 1 else ''}")
        print(f"{num_lines} line{'s' if num_lines != 1 else ''}")

    except FileNotFoundError:
        print(f"Error: File '{filename}' not found")
    except Exception as e:
        print(f"Error: {e}")

analyze_file('input.txt')
```

## Самостоятельная работа №4
Напишите программу, которая получает на вход предложение, выводит его в терминал, заменяя все запрещенные слова звездочками * (количество звездочек равно количеству букв в слове).
Запрещенные слова, разделенные символом пробела, хранятся в нижнем регистре. Программа должна заменить запрещенные слова, где бы они ни встречались, даже в середине другого слова.
Замена производится независимо от регистра: если файл input.txt содержит запрещенное слово exam, то слова exam, Exam, ExaM, EXAM и exAm должны быть заменены на ****.
Запрещенные слова:
hello email python the exam wor is
Предложение для проверки:
Hello, world! Python IS the programming language of thE future. My EMAIL is....
PYTHON is awesome!!!!
Ожидаемый результат:
*****, ***ld! ****** ** *** programming language of *** future. My ***** **....
****** ** awesome!!!!

```python
def censor_text(text, banned_words):
    banned_lower = [word.lower() for word in banned_words]

    result = []
    i = 0
    text_lower = text.lower()

    while i < len(text):
        found = False
        for banned in banned_lower:
            if text_lower.startswith(banned, i):
                result.append('*' * len(banned))
                i += len(banned)
                found = True
                break

        if not found:
            result.append(text[i])
            i += 1

    return ''.join(result)

banned_words = "hello email python the exam wor is".split()

text = """Hello, world! Python IS the programming language of thE future. My EMAIL is....
PYTHON is awesome!!!!"""

censored_text = censor_text(text, banned_words)
print(censored_text)
```
