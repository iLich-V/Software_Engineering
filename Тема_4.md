```python
        print(i[0], i[1])
```

```python

from datetime import timedelta as td # td теперь равно timedelta
# Мы воспользовались возможностью Python создавать псевдонимы

def main():
    print(
        f"Сегодня {dt.today().date()}. "
        f"День недели - {dt.today().isoweekday()}"
    )
    n = int(input('Введите количество дней: '))

    print(
        f"Через {n} дней будет {result.date()}. "
        f"День недели - {result.isoweekday()}"
    )

if __name__ == '__main__':
```

## Самостоятельная работа №1
### Дайте подробный комментарий для кода, написанного ниже. Комментарий нужен для каждой строчки кода, нужно описать что она делает.

```python
# Импорт класса datetime из модуля datetime для работы со временем
from datetime import datetime
# Импорт функции sqrt (квадратный корень) из модуля math
from math import sqrt

# Определение функции main, которая принимает произвольное количество именованных аргументов
def main(**kwargs):
    # Цикл для перебора всех переданных аргументов
    for key in kwargs.items():
        # Вычисление квадратного корня из суммы квадратов первого и второго элементов списка
        # key[1][0] - первый элемент списка-значения, key[1][1] - второй элемент
        # ** 2 означает возведение в квадрат
        result = sqrt(key[1][0] ** 2 + key[1][1] ** 2)
        # Вывод результата вычисления на экран
        print(result)

# Проверка, что скрипт запущен напрямую, а не импортирован как модуль
if __name__ =='__main__':
    # Запись текущего времени в переменную start_time
    start_time = datetime.now()
    # Вызов функции main с передачей пяти именованных аргументов
    # Каждый аргумент - список из двух чисел
    main(
        one=[10, 3],
        two=[5, 4],
        three=[15, 13],
        four=[93, 53],
        five=[133, 15]
    )
    # Вычисление разницы между текущим временем и start_time (затраченное время)
    time_costs = datetime.now() - start_time
    # Вывод времени выполнения программы в формате строки f-строкой
    print(f"Время выполнения программы - {time_costs}")
```

```python
    elif roll == 3 or roll == 4:
        dice_game()

if __name__ == '__main__':
```

```python
def triangle_area(a, b, c):
    p = (a + b + c) / 2
    area = (p * (p - a) * (p - b) * (p - c)) ** 0.5
    return area
```
